54 template<typename First, typename Last = First>
55 struct chunk_iterator {
56 using iterator = First;
57 using const_iterator = iterator;
58 using iterator_category = std::forward_iterator_tag;
59 using value_type = view_t<iterator>;
60 using reference = value_type;
61 using const_reference = value_type;
62 using difference_type = daw::iter_difference_t<iterator>;
63 using i_am_a_daw_chunk_iterator_class = void;
64
65 private:
66 iterator m_iter{ };
67 DAW_NO_UNIQUE_ADDRESS Last m_last{ };
68 difference_type m_chunk_size = 1;
69 mutable std::optional<iterator> m_next_iter{ };
70
71 [[nodiscard]] constexpr iterator find_next_iter( ) const {
72 auto result = m_iter;
73 auto n = m_chunk_size;
74 while( n > 0 and result != m_last ) {
75 --n;
76 ++result;
77 }
78 return result;
79 }
80
81 constexpr void increment( ) {
82 if( m_next_iter ) {
83 m_iter = std::move( *m_next_iter );
84 m_next_iter.reset( );
85 return;
86 }
87 m_iter = find_next_iter( );
88 }
89
90 public:
91 chunk_iterator( ) = default;
92
93 explicit constexpr chunk_iterator( iterator first )
94 : m_iter{ first }
95 , m_last{ first }
96 , m_chunk_size{ 1 } {}
97
98 explicit constexpr chunk_iterator( iterator first, Last last,
99 std::size_t chunk_size )
100 : m_iter{ first }
101 , m_last{ last }
102 , m_chunk_size{ static_cast<difference_type>( chunk_size ) } {
103 daw_ensure( m_chunk_size > 0 );
104 }
105
106 [[nodiscard]] constexpr iterator &base( ) {
107 return m_iter;
108 }
109
110 [[nodiscard]] constexpr const_iterator &base( ) const {
111 return m_iter;
112 }
113
114 [[nodiscard]] constexpr bool good( ) const {
115 return m_iter != m_last;
116 }
117
118 constexpr chunk_iterator &operator++( ) {
119 increment( );
120 return *this;
121 }
122
123 [[nodiscard]] constexpr chunk_iterator operator++( int ) {
124 auto tmp = *this;
125 increment( );
126 return tmp;
127 }
128
129 [[nodiscard]] constexpr reference operator*( ) {
130 m_next_iter = find_next_iter( );
131 return view_t<iterator>{ m_iter, *m_next_iter };
132 }
133
134 [[nodiscard]] constexpr const_reference operator*( ) const {
135 m_next_iter = find_next_iter( );
136 return view_t<iterator>{ m_iter, *m_next_iter };
137 }
138
139 [[nodiscard]] constexpr bool operator==( chunk_iterator const &rhs ) const {
140 return m_iter == rhs.m_iter;
141 }
142
143 [[nodiscard]] constexpr bool
144 operator==( chunk_iterator_end<First> const & ) const {
145 return not good( );
146 }
147
148 [[nodiscard]] constexpr bool
149 operator==( chunk_iterator_end<Last> const & ) const
150 requires( not std::same_as<First, Last> ) {
151 return not good( );
152 }
153 };
