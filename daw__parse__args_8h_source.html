206 struct Argument {
207 daw::string_view name{ };
208 daw::string_view value{ };
209
210 private:
211 friend class ::daw::Arguments;
212 friend class ::daw::args_iterator;
213
214 explicit constexpr Argument( ) = default;
215
216 explicit constexpr Argument( daw::string_view args ) {
217 if( args.starts_with( "--" ) ) {
218 // We are a named argument, starting with '--'
219 args.remove_prefix( 2 );
220 name = args.pop_front_until( '=' );
221 if( args.empty( ) ) {
222 return;
223 }
224 }
225 // We have an assigned value
226 if( args.starts_with( '"' ) and args.ends_with( '"' ) ) {
227 args.remove_prefix( 1 );
228 args.remove_suffix( 1 );
229 }
230 value = args;
231 }
232
233 public:
234 [[nodiscard]] DAW_CPP20_CX_ALLOC std::optional<std::string>
235 as_string( ) const {
236 if( not *this ) {
237 return std::nullopt;
238 }
239 return std::string( value );
240 }
241
242 private:
243 [[nodiscard]] static constexpr std::optional<std::uint64_t>
244 parse_unsigned( daw::string_view sv ) noexcept {
245 if( sv.empty( ) ) {
246 return std::nullopt;
247 }
248 std::uint64_t result = 0;
249 while( not sv.empty( ) ) {
250 auto v = as<unsigned>( as<unsigned char>( sv.pop_front( ) ) ) -
251 as<unsigned>( as<unsigned char>( '0' ) );
252 if( v >= 10U ) {
253 return std::nullopt;
254 }
255 if( result > ( max_value<std::uint64_t> - v ) / 10U ) {
256 return std::nullopt;
257 }
258 result *= 10U;
259 result += v;
260 }
261 return result;
262 }
263
264 [[nodiscard]] constexpr std::optional<std::uint64_t>
265 as_unsigned( ) const noexcept {
266 if( not *this ) {
267 return std::nullopt;
268 }
269 return parse_unsigned( value );
270 }
271
272 [[nodiscard]] constexpr std::optional<std::int64_t>
273 as_signed( ) const noexcept {
274 if( not *this ) {
275 return std::nullopt;
276 }
277 auto sv = value;
278 if( sv.empty( ) ) {
279 return std::nullopt;
280 }
281 std::int64_t const sign = sv.front( ) == '-' ? -1 : 1;
282 if( sign < 0 ) {
283 sv.remove_prefix( );
284 }
285 auto v = parse_unsigned( sv );
286 if( not v ) {
287 return std::nullopt;
288 }
289 if( *v <= as<std::uint64_t>( max_value<std::int64_t> ) ) {
290 return sign * as<std::int64_t>( *v );
291 }
292 if( *v == ( as<std::uint64_t>( max_value<std::int64_t> ) + 1U ) and
293 sign < 0 ) {
294 return min_value<std::int64_t>;
295 }
296 return std::nullopt;
297 }
298
299 explicit constexpr operator bool( ) const {
300 return value.data( ) != nullptr;
301 }
302
303 public:
304 template<typename Integer = int>
305 [[nodiscard]] constexpr std::optional<Integer> as_int( ) const noexcept {
306 static_assert( std::is_integral_v<Integer> );
307 auto const v = [this] {
308 if constexpr( std::is_signed_v<Integer> ) {
309 return as_signed( );
310 } else {
311 return as_unsigned( );
312 }
313 }( );
314 if( not v ) {
315 return std::nullopt;
316 }
317 if( daw::cxmath::cmp_less( *v, lowest_value<Integer> ) ) {
318 return std::nullopt;
319 }
320 if( daw::cxmath::cmp_greater( *v, max_value<Integer> ) ) {
321 return std::nullopt;
322 }
323 return as<Integer>( *v );
324 }
