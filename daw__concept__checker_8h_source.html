19 template<typename T>
20 DAW_CONSTEVAL bool check_movable( ) {
21 static_assert( std::is_object_v<T> );
22 static_assert( std::move_constructible<T> );
23 static_assert( std::assignable_from<T &, T> );
24 static_assert( std::swappable<T> );
