35 DAW_ATTRIB_NOINLINE std::optional<std::basic_string<CharT>>
36 read_file( daw::string_view path ) {
37 auto ec = std::error_code{ };
38 auto const fsize_tmp =
39 std::filesystem::file_size( std::string_view( path ), ec );
40 if( ec ) {
41 return std::nullopt;
42 }
43 auto const fsize = daw::narrow_cast<std::size_t>( fsize_tmp );
44 if( fsize % sizeof( CharT ) != 0 ) {
45 return std::nullopt;
46 }
47 auto element_count = fsize / sizeof( CharT ) > 0 ? fsize / sizeof( CharT )
48 : std::size_t{ 4096 };
49
50 if( daw::cmp_less( std::basic_string<CharT>{ }.max_size( ),
51 element_count ) ) {
52 // File is too big to fit into string(WIN32)
53 return std::nullopt;
54 }
55 auto const f = daw::open_cfile( path, "rb" );
56 if( not f ) {
57 return std::nullopt;
58 }
59 auto result = std::basic_string<CharT>( );
60 bool keep_going = true;
61#if defined( DAW_HAS_CPP23_STR_RESIZE_OVERWRITE )
62 while( keep_going ) {
63 auto const old_size = result.size( );
64 result.resize_and_overwrite(
65 old_size + element_count, [&]( CharT *p, std::size_t ) {
66 std::advance( p, as<std::ptrdiff_t>( old_size ) );
67 auto const num_read =
68 std::fread( p, sizeof( CharT ), element_count, f.get( ) );
69 keep_going = num_read == element_count;
70 return old_size + num_read;
71 } );
72 if( ferror( f.get( ) ) ) {
73 return std::nullopt;
74 }
75 if( keep_going ) {
76 // Ensure we don't expand too much and read efficiently
77 element_count = std::size_t{ 4096 };
78 }
79 }
80#else
81 while( keep_going ) {
82 auto const old_size = result.size( );
83 result.resize( old_size + element_count, CharT{ } );
84 CharT *p = std::next( result.data( ), as<std::ptrdiff_t>( old_size ) );
85 auto const num_read =
86 std::fread( p, sizeof( CharT ), element_count, f.get( ) );
87 if( ferror( f.get( ) ) ) {
88 return std::nullopt;
89 }
90 keep_going = num_read == element_count;
91 if( keep_going ) {
92 // Ensure we don't expand too much and read efficiently
93 element_count = std::size_t{ 4096 };
94 } else {
95 result.resize( old_size + num_read );
96 }
97 }
98#endif
99 return result;
100 }
