476 // _GLIBCXX_RESOLVE_LIB_DEFECTS
477 // 2063. Contradictory requirements for string move assignment
478 basic_string &operator=( basic_string &&_str ) noexcept(
479 Alloc_traits::_S_nothrow_move( ) ) {
480 if( !m_is_local( ) && Alloc_traits::_S_propagate_on_move_assign( ) &&
481 !Alloc_traits::_S_always_equal( ) &&
482 m_get_allocator( ) != _str.m_get_allocator( ) ) {
483 // Destroy existing storage before replacing allocator.
484 m_destroy( m_allocated_capacity );
485 m_data( m_local_data( ) );
486 m_set_length( 0 );
487 }
488 // Replace allocator if POCMA is true.
489 std::__alloc_on_move( m_get_allocator( ), _str.m_get_allocator( ) );
490
491 if( _str.m_is_local( ) ) {
492 // We've always got room for a short string, just copy it
493 // (unless this is a self-move, because that would violate the
494 // char_traits::copy precondition that the ranges don't overlap).
495 if( _builtin_expect( std::addressof( _str ) != this, true ) ) {
496 if( _str.size( ) ) {
497 this->S_copy( m_data( ), _str.m_data( ), _str.size( ) );
498 }
499 m_set_length( _str.size( ) );
500 }
501 } else if( Alloc_traits::_S_propagate_on_move_assign( ) ||
502 Alloc_traits::_S_always_equal( ) ||
503 m_get_allocator( ) == _str.m_get_allocator( ) ) {
504 // Just move the allocated pointer, our allocator can free it.
505 pointer _data = nullptr;
506 size_type _capacity;
507 if( !m_is_local( ) ) {
508 if( Alloc_traits::_S_always_equal( ) ) {
509 // _str can reuse our existing storage.
510 _data = m_data( );
511 _capacity = m_allocated_capacity;
512 } else { // _str can't use it, so free it.
513 m_destroy( m_allocated_capacity );
514 }
515 }
516
517 m_data( _str.m_data( ) );
518 m_length( _str.length( ) );
519 m_capacity( _str.m_allocated_capacity );
520 if( _data ) {
521 _str.m_data( _data );
522 _str.m_capacity( _capacity );
523 } else {
524 _str.m_data( _str.m_local_buf );
525 }
526 } else { // Need to do a deep copy
527 assign( _str );
528 }
529 _str.clear( );
530 return *this;
531 }
