250 template<typename Fn, typename Projection = std::identity>
251 struct filter_t {
252 DAW_NO_UNIQUE_ADDRESS mutable Fn m_func;
253 DAW_NO_UNIQUE_ADDRESS Projection m_projection{ };
254
255 template<Range R>
256 [[nodiscard]] constexpr auto operator( )( R &&r ) const {
257 static_assert( std::invocable<Projection, range_value_t<R>>,
258 "Projection must be invocable with range_value_t<R>" );
259 using projected_t = std::invoke_result_t<Projection, range_value_t<R>>;
260 static_assert(
261 std::invocable<Fn, projected_t>,
262 "Filter requires an invokable function with the projected value" );
263 static_assert(
264 std::convertible_to<std::invoke_result_t<Fn, projected_t>, bool>,
265 "Filter requires an invokable function that returns a bool" );
266 return filter_view<R, Fn>( DAW_FWD( r ), m_func, m_projection );
267 }
268
269 template<typename Value>
270 [[nodiscard]] constexpr auto operator( )( Value &&v ) const {
271 static_assert( std::invocable<Projection, Value>,
272 "Projection must be invocable with range_value_t<R>" );
273 using projected_t = std::invoke_result_t<Projection, Value>;
274 static_assert(
275 std::is_invocable_r_v<bool, Fn, projected_t>,
276 "Filter requires an invokable function that returns a bool" );
277 using result_t = daw::remove_cvref_t<std::invoke_result_t<Fn, Value>>;
278 if( std::invoke( m_func, std::invoke( m_projection, v ) ) ) {
279 return std::optional<result_t>( std::invoke( m_func, DAW_FWD( v ) ) );
280 }
281 return std::optional<result_t>( );
282 }
283 };
