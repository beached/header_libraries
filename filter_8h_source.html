247 template<typename Fn, typename Projection = std::identity>
248 struct filter_t {
249 DAW_NO_UNIQUE_ADDRESS mutable Fn m_func;
250 DAW_NO_UNIQUE_ADDRESS Projection m_projection{ };
251
252 template<Range R>
253 [[nodiscard]] constexpr auto operator( )( R &&r ) const {
254 static_assert( std::invocable<Projection, range_value_t<R>>,
255 "Projection must be invocable with range_value_t<R>" );
256 using projected_t = std::invoke_result_t<Projection, range_value_t<R>>;
257 static_assert(
258 std::invocable<Fn, projected_t>,
259 "Filter requires an invokable function with the projected value" );
260 static_assert(
261 std::convertible_to<std::invoke_result_t<Fn, projected_t>, bool>,
262 "Filter requires an invokable function that returns a bool" );
263 return filter_view<R, Fn>( DAW_FWD( r ), m_func, m_projection );
264 }
265
266 template<typename Value>
267 [[nodiscard]] constexpr auto operator( )( Value &&v ) const {
268 static_assert( std::invocable<Projection, Value>,
269 "Projection must be invocable with range_value_t<R>" );
270 using projected_t = std::invoke_result_t<Projection, Value>;
271 static_assert(
272 std::is_invocable_r_v<bool, Fn, projected_t>,
273 "Filter requires an invokable function that returns a bool" );
274 using result_t = daw::remove_cvref_t<std::invoke_result_t<Fn, Value>>;
275 if( std::invoke( m_func, std::invoke( m_projection, v ) ) ) {
276 return std::optional<result_t>( std::invoke( m_func, DAW_FWD( v ) ) );
277 }
278 return std::optional<result_t>( );
279 }
280 };
