25 template<Range R>
26 struct drop_view : private maybe_owning_range<R> {
27 using base_t = maybe_owning_range<R>;
28 using iterator = typename base_t::iterator;
29 using const_iterator = typename base_t::const_iterator;
30 using iterator_last = typename base_t::iterator_last;

32 using difference_type = daw::range_difference_t<R>;
33 using daw_i_am_a_drop_view_class = void;
34
35 private:
36 // How many leading elements are dropped. An offset, unlike a cached
37 // iterator, stays valid when the view is copied or moved.
38 difference_type m_dropped = 0;
39
40 [[nodiscard]] static constexpr difference_type
41 drop_while( iterator first, iterator_last last, auto &&fn ) {
42 difference_type n = 0;
43 while( first != last ) {
44 if( not std::invoke( fn, as_const( *first ) ) ) {
45 break;
46 }
47 ++first;
48 ++n;
49 }
50 return n;
51 }
52
53 [[nodiscard]] static constexpr difference_type
54 drop_until( iterator first, iterator_last last, auto &&fn ) {
55 difference_type n = 0;
56 while( first != last ) {
57 if( std::invoke( fn, as_const( *first ) ) ) {
58 break;
59 }
60 ++first;
61 ++n;
62 }
63 return n;
64 }

66 public:
67 drop_view( ) = default;

69 explicit constexpr drop_view(
70 daw::constructible<base_t> auto &&r, drop_while_t,
71 InvocableAs<bool( range_const_reference_t<R> )> auto &&fn )
