235 template<BidirectionalRange R>
236 [[nodiscard]] DAW_CPP23_STATIC_CALL_OP constexpr daw::remove_rvalue_ref_t<R>
237 operator( )( R &&r ) DAW_CPP23_STATIC_CALL_OP_CONST {
238 auto first = std::begin( r );
239 auto const last = std::end( r );
240 auto it = safe_move_next( first, last );
241 auto tail = it;
242
243 if constexpr( RandomRange<R> ) {
244 if( first != last ) {
245 --tail;
246 while( first < tail ) {
247 std::iter_swap( first, tail );
248 ++first;
249 --tail;
250 }
251 }
252 } else {
253 while( true ) {
254 if( first == tail || first == --tail ) {
255 break;
256 }
257 std::iter_swap( first, tail );
258 ++first;
259 }
260 }
261 return DAW_FWD( r );
262 }
