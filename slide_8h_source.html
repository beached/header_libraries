57 template<typename First, typename Last = First>
58 struct slide_iterator {
59 using iterator = First;
60 using const_iterator = iterator;
61 using iterator_category = std::forward_iterator_tag;
62 using value_type = view_t<iterator, Last>;
63 using reference = value_type;
64 using const_reference = value_type;
65 using difference_type = daw::iter_difference_t<iterator>;
66 using i_am_a_daw_slide_iterator_class = void;
67
68 private:
69 iterator m_iter{ };
70 difference_type m_slide_size = 1;
71 DAW_NO_UNIQUE_ADDRESS Last m_last{ };
72 mutable iterator m_next_iter = m_iter;
73
74 public:
75 explicit slide_iterator( ) = default;
76
77 explicit constexpr slide_iterator( iterator first, Last last )
78 : m_iter{ std::move( first ) }
79 , m_last{ std::move( last ) }
80 , m_next_iter( safe_move_next( m_iter, m_last, m_slide_size ) ) {}
81
82 explicit constexpr slide_iterator( iterator first, Last last,
83 std::size_t slide_size )
84 : m_iter{ std::move( first ) }
85 , m_slide_size( as<difference_type>( slide_size ) )
86 , m_last{ std::move( last ) }
87 , m_next_iter( safe_move_next( m_iter, m_last, m_slide_size ) ) {
88 daw_ensure( m_slide_size > 0 );
89 }
90
91 [[nodiscard]] constexpr iterator &base( ) {
92 return m_iter;
93 }
94
95 [[nodiscard]] constexpr const_iterator &base( ) const {
96 return m_iter;
97 }
98
99 constexpr slide_iterator &operator++( ) {
100 if( m_next_iter == m_last ) {
101 m_iter = m_last;
102 return *this;
103 }
104 ++m_iter;
105 ++m_next_iter;
106 return *this;
107 }
108
109 [[nodiscard]] constexpr slide_iterator operator++( int ) {
110 auto tmp = *this;
111 this->operator++( );
112 return tmp;
113 }
114
115 [[nodiscard]] constexpr reference operator*( ) {
116 return view_t<iterator>{ m_iter, m_next_iter };
117 }
118
119 [[nodiscard]] constexpr const_reference operator*( ) const {
120 return view_t<iterator>{ m_iter, m_next_iter };
121 }
122
123 [[nodiscard]] constexpr bool operator==( slide_iterator const &rhs ) const {
124 return m_iter == rhs.m_iter and m_slide_size == rhs.m_slide_size;
125 }
126
127 [[nodiscard]] constexpr bool
128 operator==( slide_iterator_end<First> const & ) const {

137 template<typename First, typename Last = First>
138 struct slide_index_iterator {
139 using iterator = First;
140 using const_iterator = iterator;
141 using iterator_category = std::forward_iterator_tag;
142 using value_type = view_t<iterator>;
143 using reference = value_type;
144 using const_reference = value_type;
145 using difference_type = daw::iter_difference_t<iterator>;
146 using i_am_a_daw_slide_iterator_class = void;
147
148 private:
149 iterator m_iter{ };
150 difference_type m_count = 0; // windows left
151 difference_type m_window = 0; // elements in each window
152
153 public:
154 slide_index_iterator( ) = default;
155
156 explicit constexpr slide_index_iterator( iterator first, Last last )
157 : slide_index_iterator( std::move( first ), std::move( last ),
158 std::size_t{ 1 } ) {}
159
160 explicit constexpr slide_index_iterator( iterator first, Last last,
161 std::size_t slide_size )
162 : m_iter{ std::move( first ) } {
163 daw_ensure( slide_size > 0 );
164 auto const n = as<difference_type>( slide_size );
165 auto const len = counted_length( m_iter, last );
166 m_window = std::min( n, len );
167 // a range shorter than the window gets one short window, an empty
168 // range gets none
169 m_count = len == 0 ? 0 : ( len >= n ? len - n + 1 : 1 );
170 }
171
172 [[nodiscard]] constexpr iterator &base( ) {
173 return m_iter;
174 }
175
176 [[nodiscard]] constexpr const_iterator &base( ) const {
177 return m_iter;
178 }
179
180 constexpr slide_index_iterator &operator++( ) {
181 ++m_iter;
182 --m_count;
183 return *this;
184 }
185
186 [[nodiscard]] constexpr slide_index_iterator operator++( int ) {
187 auto tmp = *this;
188 this->operator++( );
189 return tmp;
190 }
191
192 [[nodiscard]] constexpr reference operator*( ) {
193 return view_t<iterator>{ m_iter, m_iter + m_window };
194 }
195
196 [[nodiscard]] constexpr const_reference operator*( ) const {
197 return view_t<iterator>{ m_iter, m_iter + m_window };
198 }
199
200 [[nodiscard]] constexpr bool
201 operator==( slide_index_iterator const &rhs ) const {
202 return ( m_count <= 0 and rhs.m_count <= 0 ) or m_iter == rhs.m_iter;
203 }
204
205 [[nodiscard]] constexpr bool
206 operator==( slide_iterator_end<First> const & ) const {
207 return m_count <= 0;
208 }
209 };
