78 struct concat_iterator {
79 using iterator_category = std::forward_iterator_tag;
80
81 using types_t =
82 std::tuple<daw::remove_cvrvref_t<daw::remove_rvalue_ref_t<Ranges>>...>;
83 using value_type =
84 std::common_type_t<range_value_t<daw::remove_rvalue_ref_t<Ranges>>...>;
85 using reference = std::common_reference_t<
86 range_reference_t<daw::remove_rvalue_ref_t<Ranges>>...>;
87 using const_reference = std::common_reference_t<
88 range_const_reference_t<daw::remove_rvalue_ref_t<Ranges>>...>;
89 using difference_type = widest_type_t<daw::range_difference_t<Ranges>...>;
90 using i_am_a_daw_concat_iterator_class = void;
91
92 private:
93 using last_iterator =
94 pimpl::concat_iterator_end<daw::iterator_end_t<Ranges>...>;
95 using last_const_iterator =
96 pimpl::concat_iterator_end<daw::const_iterator_end_t<Ranges>...>;
97 using position_t = variant_range_storage_t<
98 view_t<iterator_t<Ranges>, iterator_end_t<Ranges>>...>;
99
100 using cv_t = CV;
101 using cv_pointer_t = cv_t *;
102 using cv_const_pointer_t = cv_t const *;
103 using positions_t = std::array<position_t, cv_t::range_size>;
104
105 cv_pointer_t m_view{ };
106 positions_t m_positions = positions_t{ };
107 std::size_t m_index = cv_t::range_size;
108
109 template<std::size_t... Is>
110 [[nodiscard]] static constexpr positions_t
111 make_positions( cv_const_pointer_t view, std::index_sequence<Is...> ) {
112 return positions_t{
113 position_t( view_t{ view->m_storage[Is].template get<Is>( ).rbegin( ),
114 view->m_storage[Is].template get<Is>( ).rend( ) },
115 daw::constant_v<Is> )... };
116 }
117
118 constexpr void skip_empty_ranges( ) {
119 while( m_index < cv_t::range_size and
120 m_positions[m_index].visit( []( auto const &r ) -> bool {
121 return r.rbegin( ) == r.rend( );
122 } ) ) {
123 ++m_index;
124 }
125 }
126
127 public:
128 explicit constexpr concat_iterator( cv_pointer_t view )
129 : m_view( view )
130 , m_positions( make_positions(
131 view, std::make_index_sequence<cv_t::range_size>{ } ) )
132 , m_index( 0 ) {
133 skip_empty_ranges( );
134 }
135
136 concat_iterator( ) = default;
137
138 constexpr concat_iterator &operator++( ) {
139 m_positions[m_index].visit( []( auto &p ) {
140 (void)++p;
141 } );
142 skip_empty_ranges( );
143 return *this;
144 }
145
146 [[nodiscard]] constexpr concat_iterator operator++( int ) {
147 auto tmp = *this;
148 ++( *this );
149 return tmp;
150 }
151
152 [[nodiscard]] constexpr bool good( ) const noexcept {
153 return m_index != cv_t::range_size;
154 }
155
156 [[nodiscard]] constexpr reference operator*( ) {
157 return m_positions[m_index].visit( []( auto &p ) -> reference {
158 return *p;
159 } );
160 }
161
162 [[nodiscard]] constexpr const_reference operator*( ) const {
163 return m_positions[m_index].visit(
164 []( auto const &p ) -> const_reference {
165 return *p;
166 } );
167 }
168
169 [[nodiscard]] constexpr bool
170 operator==( concat_iterator const &rhs ) const noexcept {
171 return m_view == rhs.m_view and m_index == rhs.m_index and
172 m_positions == rhs.m_positions;
173 }
174
175 [[nodiscard]] constexpr bool
176 operator==( last_iterator const & ) const noexcept {
177 return not good( );
178 }
179 };

279 struct Concat_t {
280 template<typename R, ForwardRange... Rs>
281 [[nodiscard]] DAW_CPP23_STATIC_CALL_OP constexpr auto
282 operator( )( R &&r, Rs &&...rs ) DAW_CPP23_STATIC_CALL_OP_CONST {
283 using std::get;
284 if constexpr( sizeof...( Rs ) == 0 and requires {
285 typename daw::remove_cvref_t<
286 R>::i_am_a_daw_zip_view_class;
287 } ) {
288 return [&]<std::size_t... Is>( std::index_sequence<Is...> ) {
289 return concat_view{ std::get<Is>( r.m_ranges ).get( )... };
290 }( std::make_index_sequence<daw::remove_cvref_t<R>::range_count>{ } );
291 } else if constexpr( ( sizeof...( Rs ) == 0 ) and
292 daw::is_tuple_like_v<daw::remove_cvref_t<R>> and
293 pimpl::is_tuple_of_fwd_ranges<
294 daw::remove_cvref_t<R>>( ) ) {
295 static_assert( sizeof...( Rs ) == 0 );
296 return [&]<std::size_t... Is>( std::index_sequence<Is...> ) {
297 static_assert(
298 ( Range<std::tuple_element_t<Is, daw::remove_cvref_t<R>>> and
299 ... ),
300 "Concat when passed a tuple expects a tuple of Ranges" );
301 return concat_view{ std::get<Is>( DAW_FWD( r ) )... };
302 }( std::make_index_sequence<
303 std::tuple_size_v<daw::remove_cvref_t<R>>>{ } );
304 } else {
305 static_assert( ForwardRange<R> );
306 return concat_view<R, Rs...>{ DAW_FWD( r ), DAW_FWD( rs )... };
307 }
308 }
309 };
