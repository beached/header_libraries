23 template<typename Result = std::ptrdiff_t>
24 [[nodiscard]] constexpr Result ranges_distance( Iterator auto first,
25 Iterator auto const &last ) {
26 if constexpr( requires { last - first; } ) {
27 return as<Result>( last - first );
28 } else {
29 Result d = Result{ };
30 while( first != last ) {
31 ++first;
32 ++d;
33 }
34 return d;
35 }
36 }

39 [[nodiscard]] constexpr Result ranges_distance( Range auto const &r ) {
