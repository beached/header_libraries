191 template<std::size_t... Is>
192 explicit constexpr zip_iterator( ZR *zr, std::index_sequence<Is...> )
193 : m_parent( zr )
194 , m_iters{ std::begin( std::get<Is>( zr->m_ranges ).get( ) )... } {
195 if constexpr( counted_v ) {
196 auto const len = std::min( { pimpl::counted_length(
197 std::get<Is>( m_iters ),
198 std::end( std::get<Is>( zr->m_ranges ).get( ) ) )... } );
199 // An unbounded length must saturate, not truncate, when
200 // difference_type is narrower than std::ptrdiff_t
201 constexpr auto max_len = daw::max_value<difference_type>;
202 m_remaining.value = daw::cmp_less( max_len, len )
203 ? max_len
204 : static_cast<difference_type>( len );
205 }
