22 template<typename R>
23 struct skip_view : private maybe_owning_range<R> {
24 using base_t = maybe_owning_range<R>;
25 using iterator = typename base_t::iterator;
26 using const_iterator = typename base_t::const_iterator;
27 using iterator_last = typename base_t::iterator_last;
28 using const_iterator_last = typename base_t::const_iterator_last;
29 using daw_i_am_a_skip_view_class = void;
30
31 private:
32 using difference_type = daw::range_difference_t<R>;
33 // How many leading elements are skipped. An offset, unlike a cached
34 // iterator, stays valid when the view is copied or moved.
35 difference_type m_skipped = 0;
36
37 public:
38 skip_view( ) = default;
39
40 explicit constexpr skip_view( daw::constructible<base_t> auto &&r,
41 std::size_t how_many )
42 : base_t( DAW_FWD( r ) ) {
43 auto first = base_t::begin( );
44 auto const last = base_t::end( );
45 if constexpr( RandomIterator<iterator> ) {
46 auto const range_size = pimpl::ranges_distance( first, last );
47 m_skipped = std::min( { as<difference_type>( how_many ), range_size } );
48 } else {
49 // Input
50 for( ; how_many > 0 and first != last; --how_many ) {
51 ++first;
52 ++m_skipped;
53 }
54 }
55 }
56
57 [[nodiscard]] constexpr iterator begin( ) {
58 return std::next( base_t::begin( ), m_skipped );
59 }
60
61 [[nodiscard]] constexpr const_iterator begin( ) const {
62 return std::next( base_t::begin( ), m_skipped );
63 }
64
65 [[nodiscard]] constexpr iterator_last end( ) {
66 return base_t::end( );
67 }
68
69 [[nodiscard]] constexpr const_iterator_last end( ) const {
70 return base_t::end( );
71 }
72
73 [[nodiscard]] constexpr bool
74 operator==( skip_view const & ) const = default;
75 };
