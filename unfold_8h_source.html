62 template<typename State, typename Fn>
63 struct unfold_iterator {
64 using step_type = std::invoke_result_t<Fn const &, State const &>;
65 using iterator_category = std::input_iterator_tag;
66 using value_type = typename step_type::value_type::first_type;
67 using reference = value_type const &;
68 using pointer = value_type const *;
69 using difference_type = std::ptrdiff_t;
70
71 private:
72 Fn const *m_fn = nullptr;
73 step_type m_step{ };
74
75 public:
76 unfold_iterator( ) = default;
77
78 explicit constexpr unfold_iterator( Fn const &fn, State const &seed )
79 : m_fn( std::addressof( fn ) )
80 , m_step( std::invoke( fn, seed ) ) {}
81
82 [[nodiscard]] constexpr reference operator*( ) const {
83 return m_step->first;
84 }
85
86 [[nodiscard]] constexpr pointer operator->( ) const {
87 return std::addressof( m_step->first );
88 }
89
90 constexpr unfold_iterator &operator++( ) {
91 // Move the next state out before overwriting the step that holds it
92 State next = std::move( m_step->second );
93 m_step = std::invoke( *m_fn, std::as_const( next ) );
94 return *this;
95 }
96
97 constexpr unfold_iterator operator++( int )
98 requires( std::copy_constructible<unfold_iterator> )
99 {
100 auto result = *this;
101 operator++( );
102 return result;
103 }
104
105 constexpr void operator++( int )
106 requires( not std::copy_constructible<unfold_iterator> )
107 {
108 operator++( );
109 }
110
113 [[nodiscard]] constexpr bool
114 operator==( unfold_iterator const &rhs ) const {
115 if constexpr( std::equality_comparable<step_type> ) {
116 return m_step == rhs.m_step;
117 } else {
118 return not m_step and not rhs.m_step;
119 }
120 }

122 [[nodiscard]] constexpr bool
123 operator==( unfold_iterator_end<value_type> const & ) const {
124 return not m_step.has_value( );
