43 template<typename Result>
44 constexpr Result parse_int( daw::string_view &str ) {
45 intmax_t count = std::numeric_limits<Result>::digits;
46 Result result = 0;
47 bool is_neg = false;
48 if( '-' == str.front( ) ) {
49
50 daw::exception::precondition_check<invalid_input_exception>(
51 std::numeric_limits<Result>::is_signed );
52
53 is_neg = true;
54 str.remove_prefix( );
55 }
56 while( count > 0 and !str.empty( ) and is_number( str.front( ) ) ) {
57 result *= static_cast<Result>( 10 );
58 result += static_cast<Result>( str.front( ) - '0' );
59 --count;
60 str.remove_prefix( );
61 }
62
63 daw::exception::precondition_check<numeric_overflow_exception>(
64 str.empty( ) );
65 if( is_neg ) {
66 result *= static_cast<Result>( -1 );
67 }
68 return result;
69 }

119 constexpr daw::string_view
120 parse_to_value( daw::string_view str DAW_LIFETIME_BOUND,
121 tag_t<daw::string_view> ) {
122 daw::exception::precondition_check<empty_input_exception>(
123 not str.empty( ) );
124
125 struct input_too_small_exception {};
126 struct missing_expected_quotes_exception {};
127
128 daw::exception::precondition_check<input_too_small_exception>(
129 str.size( ) > 2 );
130 daw::exception::precondition_check<missing_expected_quotes_exception>(
131 str.front( ) == '"' );
132
133 str.remove_prefix( );
134 auto const first = str.data( );
135 char last_char = str.pop_front( );
136 while( not str.empty( ) and
137 not impl::is_quote( last_char, str.front( ) ) ) {
138 last_char = str.pop_front( );
139 }
140 daw::exception::precondition_check<missing_expected_quotes_exception>(
141 str.data( ) != nullptr and str.front( ) == '"' );
142
143 return daw::string_view( first, str.data( ) );
144 }

204 struct default_splitter {
205 daw::string_view m_delemiter;
206
207 constexpr auto operator( )( daw::string_view str ) const noexcept {
208 using sv_size_t = typename daw::string_view::size_type;
209 struct result_t {
210 sv_size_t first;
211 sv_size_t last;
212 };
213 auto const pos = str.find( m_delemiter );
214 if( pos < str.size( ) ) {
215 return result_t{ pos, pos + m_delemiter.size( ) };
216 }
217 return result_t{ daw::string_view::npos, daw::string_view::npos };
218 }
219 };

224 template<bool skip_multiple>
225 struct basic_whitespace_splitter {
226 constexpr auto operator( )( daw::string_view str ) const {
227 struct result_t {
228 typename daw::string_view::size_type first;
229 typename daw::string_view::size_type last;
230 };
231
232 using sv_size_t = typename daw::string_view::size_type;
233 sv_size_t n = 0;
234 size_t const sz = str.size( );
235
236 while( n < sz and not_unicode_whitespace( str[n] ) ) {
237 ++n;
238 }
239 if( n == sz ) {
240 return result_t{ daw::string_view::npos, daw::string_view::npos };
241 }
242 if( !skip_multiple ) {
243 return result_t{ n, n + 1 };
244 }
245 auto const f = n;
246 while( n < sz and is_unicode_whitespace( str[n] ) ) {
247 ++n;
248 }
249 return result_t{ f, n };
250 }
251 };
