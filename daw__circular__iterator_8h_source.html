66 constexpr difference_type get_offset( difference_type n ) const noexcept {
67 auto const sz =
68 static_cast<difference_type>( impl::container_size( *m_container ) );
69 n += m_position;
70 if( n < 0 ) {
71 n *= -1;
72 n %= sz;
73 return sz - n;
74 } else if( n >= sz ) {
75 return n % sz;
76 }
77 return n;
78 }
79
80 constexpr decltype( auto ) get_iterator( difference_type n ) const {
81 auto result = std::next( std::begin( *m_container ), n );
82 return result;
83 }
