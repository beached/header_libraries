76 template<ForwardIterator First, typename IteratorLast, typename Delimiter,
77 typename Projection>
78 struct split_iterator {
79 using iterator_category = std::forward_iterator_tag;
80 using value_type = view_t<First, First>;
81 using reference = value_type;
82 using const_reference = value_type;
83 using difference_type = daw::iter_difference_t<First>;
84 using i_am_a_daw_split_iterator_class = void;
85
86 private:
87 First m_first = First{ };
88 IteratorLast m_last = IteratorLast{ };
89 DAW_NO_UNIQUE_ADDRESS Delimiter m_delimiter{ };
90 DAW_NO_UNIQUE_ADDRESS Projection m_projection{ };
91 mutable std::optional<First> m_next = std::nullopt;
92
93 constexpr void increment_iterator( ) const {
94 if( m_next or m_first == m_last ) {
95 return;
96 }
97 if constexpr( std::is_invocable_v<Delimiter, iter_reference_t<First>> ) {
98 m_next = daw::pipelines::FindIf(
99 view_t{ m_first, m_last }, m_delimiter, m_projection );
100 } else if constexpr( Range<Delimiter> ) {
101 m_next = std::search(
102 m_first, m_last, std::begin( m_delimiter ), std::end( m_delimiter ) );
103 } else {
104 m_next = daw::pipelines::Find(
105 view_t{ m_first, m_last }, m_delimiter, m_projection );
106 }
107 assert( m_next );
108 }
109
110 constexpr auto get_delimiter_length( ) const {
111 return pimpl::delimiter_length_t<Delimiter>{ m_delimiter }.size( );
112 }
113
114 public:
115 split_iterator( ) = default;
116
117 explicit constexpr split_iterator( First first, IteratorLast last,
118 Delimiter d, Projection p )
119 : m_first( std::move( first ) )
120 , m_last( std::move( last ) )
121 , m_delimiter( std::move( d ) )
122 , m_projection( std::move( p ) ) {
123 increment_iterator( );
124 }
125
126 constexpr split_iterator &operator++( ) {
127 increment_iterator( );
128 assert( m_next );
129 m_first = std::move( *m_next );
130 m_next.reset( );
131 if( m_first != m_last ) {
132 std::advance( m_first, get_delimiter_length( ) );
133 }
134 return *this;
135 }
136
137 [[nodiscard]] constexpr split_iterator operator++( int ) {
138 auto result = *this;
139 operator++( );
140 return result;
141 }
142
143 [[nodiscard]] constexpr const_reference operator*( ) const {
144 increment_iterator( );
145 assert( m_next );
146 return const_reference{ m_first, *m_next };
147 }
148
149 [[nodiscard]] constexpr bool good( ) const {
150 return m_first != m_last;
151 }
152
153 [[nodiscard]] constexpr bool operator==( split_iterator const &rhs ) const {
154 return m_first == rhs.m_first and m_last == rhs.m_last and
155 m_next == rhs.m_next;
156 }
157
158 [[nodiscard]] constexpr bool
159 operator==( split_iterator_end<First> const & ) const {
160 return not good( );
161 }
162 };

181 DAW_NO_UNIQUE_ADDRESS Delimiter m_delimiter{ };
182 DAW_NO_UNIQUE_ADDRESS Projection m_projection = Projection{ };

184 static constexpr auto advance_to_last( auto f, auto l ) {
185 if constexpr( std::same_as<decltype( f ), decltype( l )> ) {
186 // common range, the end is already an iterator
187 return l;
188 } else if constexpr( RandomRange<R> ) {
189 auto const sz = pimpl::ranges_distance( f, l );
190 return std::next( f, sz );
191 } else {
192 while( f != l ) {
193 ++f;
194 }
195 return f;
196 }
197 }
198
199 // Computed on demand instead of cached at construction, so a copied or
200 // moved split_view never holds an iterator into another object.
201 [[nodiscard]] constexpr auto last( ) {
202 return advance_to_last( base_t::begin( ), base_t::end( ) );
203 }
204
205 [[nodiscard]] constexpr auto last( ) const {
206 return advance_to_last( base_t::begin( ), base_t::end( ) );
207 }
208
209 public:
210 split_view( ) = default;

212 constexpr split_view( constructible<base_t> auto &&r, Delimiter d,
213 Projection p = Projection{ } )
214 : base_t{ DAW_FWD( r ) }
215 , m_delimiter{ std::move( d ) }
216 , m_projection{ std::move( p ) } {}

218 [[nodiscard]] constexpr iterator begin( ) {

237 return const_iterator_last{ };
238 }

241 return const_iterator_last{ };
242 }
