335 template<typename T, std::size_t sz = sizeof( T ),
336 bool is_integral = daw::is_integral_v<T>,
337 bool is_signed = daw::is_signed_v<T>>
338 DAW_CONSTEVAL auto next_wider_helper( ) {
339 if constexpr( not is_integral ) {
340 return type_identity<T>{ };
341 } else if constexpr( sz == 1 ) {
342 if constexpr( is_signed ) {
343 return type_identity<std::int16_t>{ };
344 } else {
345 return type_identity<std::uint16_t>{ };
346 }
347 } else if constexpr( sz == 2 ) {
348 if constexpr( is_signed ) {
349 return type_identity<std::int32_t>{ };
350 } else {
351 return type_identity<std::uint32_t>{ };
352 }
353 } else if constexpr( sz == 4 ) {
354 if constexpr( is_signed ) {
355 return type_identity<std::int64_t>{ };
356 } else {
357 return type_identity<std::uint64_t>{ };
358 }
359 } else if constexpr( sz == 8 ) {
360#if defined( DAW_HAS_INT128 )
361 if constexpr( is_signed ) {
362 return type_identity<daw::int128_t>{ };
363 } else {
364 return type_identity<daw::uint128_t>{ };
365 }
366#else
367 return type_identity<T>{ };
368#endif
369 } else {
370 return type_identity<T>{ };
371 }
372 }
373 } // namespace next_wider_impl
