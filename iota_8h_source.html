26 template<typename T>
27 struct iota_iterator {
28 using iterator_category = std::random_access_iterator_tag;
29 using value_type = T;
30 using reference = value_type;
31 using const_reference = value_type;
32 using difference_type = daw::next_wider_t<daw::make_signed_t<T>>;
33 using size_type = daw::next_wider_t<daw::make_unsigned_t<T>>;
34
35 private:
36 T m_value = max_value<T>;
37
38 public:
39 iota_iterator( ) = default;
40
41 explicit constexpr iota_iterator(
42 daw::explicitly_convertible_to<T> auto &&value )
43 : m_value( as<T>( DAW_FWD( value ) ) ) {}
44
45 [[nodiscard]] DAW_ATTRIB_INLINE constexpr reference operator*( ) const {
46 return m_value;
47 }
48
49 constexpr iota_iterator &operator++( ) {
50 return *this += value_type{ 1 };
51 }
52
53 [[nodiscard]] constexpr iota_iterator operator++( int ) {
54 iota_iterator result = *this;
55 *this += value_type{ 1 };
56 return result;
57 }
58
59 constexpr iota_iterator &operator--( ) {
60 return *this -= value_type{ 1 };
61 }
62
63 [[nodiscard]] constexpr iota_iterator operator--( int ) {
64 iota_iterator result = *this;
65 *this -= value_type{ 1 };
66 return result;
67 }
68
69 DAW_ATTRIB_INLINE constexpr iota_iterator &operator+=( difference_type n ) {
70 // size_type is the unsigned type as wide as difference_type. Adding in
71 // it wraps, where a signed sum could overflow for a large step
72 m_value = as<T>( as<size_type>( m_value ) + as<size_type>( n ) );
73 return *this;
74 }
75
76 DAW_ATTRIB_INLINE constexpr iota_iterator &operator-=( difference_type n ) {
77 m_value = as<T>( as<size_type>( m_value ) - as<size_type>( n ) );
78 return *this;
79 }
80
81 [[nodiscard]] constexpr friend iota_iterator
82 operator+( iota_iterator lhs, difference_type n ) noexcept {
83 lhs += n;
84 return lhs;
85 }
86
87 [[nodiscard]] constexpr friend iota_iterator
88 operator+( difference_type n, iota_iterator rhs ) noexcept {
89 rhs += n;
90 return rhs;
91 }
92
93 [[nodiscard]] constexpr friend iota_iterator
94 operator-( iota_iterator lhs, difference_type n ) noexcept {
95 lhs -= n;
96 return lhs;
97 }
98
99 [[nodiscard]] constexpr difference_type
100 operator-( iota_iterator const &i ) const {
101 return as<difference_type>( m_value ) - as<difference_type>( i.m_value );
102 }
103
104 DAW_ATTRIB_FLATTEN [[nodiscard]] constexpr reference
105 operator[]( difference_type n ) const noexcept {
106 return *( *this + n );
107 }
108
109 [[nodiscard]] constexpr bool
110 operator==( iota_iterator const &rhs ) const = default;
111
112 // clang-format off
113 [[nodiscard]] friend constexpr auto operator<=>
114 ( iota_iterator const &lhs, iota_iterator const &rhs ) {
115 return lhs.m_value <=> rhs.m_value;
116 }
117 // clang-format on
118 };

121 struct iota_view {
122 using iterator = iota_iterator<T>;
123 using difference_type = daw::iter_difference_t<iterator>;
124 using value_type = daw::iter_value_t<iterator>;
125 using size_type = daw::iter_size_t<iterator>;
126
127 private:
128 T m_first = { };
129 T m_last = max_value<T>;
130
131 public:
132 explicit iota_view( ) = default;
133
134 explicit constexpr iota_view(
135 daw::explicitly_convertible_to<T> auto &&last )
136 : m_last{ as<T>( DAW_FWD( last ) ) } {
137 daw_dbg_ensure( m_first <= m_last );
138 }
139
140 explicit constexpr iota_view(
141 daw::explicitly_convertible_to<T> auto &&first,
142 daw::explicitly_convertible_to<T> auto &&last )
143 : m_first{ as<T>( DAW_FWD( first ) ) }
144 , m_last{ as<T>( DAW_FWD( last ) ) } {
145 daw_dbg_ensure( m_first <= m_last );
146 }
147
148 [[nodiscard]] constexpr iterator begin( ) const {
149 return iterator{ m_first };
150 }
151
152 [[nodiscard]] constexpr iterator end( ) const {
153 return iterator{ m_last };
154 }
155
156 [[nodiscard]] constexpr bool empty( ) const {
157 return m_first == m_last;
158 }
159
160 DAW_ATTRIB_FLATTEN [[nodiscard]] constexpr value_type
161 operator[]( difference_type n ) const noexcept {
162 return begin( )[n];
163 }
164
165 [[nodiscard]] constexpr bool operator==( iota_view const &rhs ) const {
166 return m_first == rhs.m_first and m_last == rhs.m_last;
167 }
