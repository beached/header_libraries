56 template<typename First, typename Last = First>
57 struct flatten_iterator {
58 using iterator_t = First;
59 using sub_iterator_t = daw::iterator_t<daw::iter_value_t<iterator_t>>;
60 using iterator_category = std::forward_iterator_tag;
61 using value_type = daw::iter_value_t<sub_iterator_t>;
62 using reference = daw::iter_reference_t<sub_iterator_t>;
63 using const_reference = daw::iter_const_reference_t<sub_iterator_t>;
64 using difference_type = daw::iter_difference_t<First>;
65 using i_am_a_daw_flatten_iterator_class = void;
66
67 private:
68 iterator_t m_iter{ };
69 DAW_NO_UNIQUE_ADDRESS Last m_last{ };
70 sub_iterator_t m_cur_first{ };
71
72 DAW_ATTRIB_INLINE constexpr void advance_to_non_empty_range( ) {
73 while( m_iter != m_last ) {
74 m_cur_first = std::begin( *m_iter );
75 if( m_cur_first != std::end( *m_iter ) ) {
76 return;
77 }
78 ++m_iter;
79 }
80 m_cur_first = sub_iterator_t{ };
81 }
82
83 DAW_ATTRIB_INLINE constexpr void inc_range( ) {
84 assert( m_iter != m_last );
85 ++m_iter;
86 if( m_iter == m_last ) {
87 m_cur_first = sub_iterator_t{ };
88 } else {
89 m_cur_first = std::begin( *m_iter );
90 }
91 }
92
93 DAW_ATTRIB_INLINE constexpr void inc_sub_range( ) {
94 assert( m_cur_first != sub_iterator_t{ } );
95 assert( m_cur_first != std::end( *m_iter ) );
96 ++m_cur_first;
97 if( m_cur_first == std::end( *m_iter ) ) {
98 ++m_iter;
99 advance_to_non_empty_range( );
100 }
101 }
102
103 public:
104 explicit flatten_iterator( ) = default;
105
106 explicit constexpr flatten_iterator( First first, Last last )
107 : m_iter( first )
108 , m_last( last )
109 , m_cur_first( m_iter == m_last ? sub_iterator_t{ }
110 : std::begin( *m_iter ) ) {
111 advance_to_non_empty_range( );
112 }
113
114 [[nodiscard]] constexpr bool good( ) const {
115 return m_iter != m_last;
116 }
117
118 constexpr flatten_iterator &operator++( ) {
119 inc_sub_range( );
120 return *this;
121 }
122
123 [[nodiscard]] constexpr flatten_iterator operator++( int ) {
124 auto tmp = *this;
125 ++( *this );
126 return tmp;
127 }
128
129 [[nodiscard]] constexpr reference operator*( ) noexcept {
130 assert( m_iter != m_last );
131 assert( m_cur_first != std::end( *m_iter ) );
132 assert( m_cur_first != sub_iterator_t{ } );
133 return *m_cur_first;
134 }
135
136 [[nodiscard]] constexpr const_reference operator*( ) const noexcept {
137 assert( m_iter != m_last );
138 assert( m_cur_first != std::end( *m_iter ) );
139 assert( m_cur_first != sub_iterator_t{ } );
140 return *m_cur_first;
141 }
142
143 [[nodiscard]] constexpr bool
144 operator==( flatten_iterator const &rhs ) const {
145 return m_iter == rhs.m_iter and m_cur_first == rhs.m_cur_first;
146 }
147
148 [[nodiscard]] constexpr bool
149 operator==( flatten_iterator_end<First> const & ) const {
150 return not good( );
151 }
152 };
