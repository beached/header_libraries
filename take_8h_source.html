34 template<Range R>
35 struct take_view : private pimpl::stored_range_base_t<R> {
36 using base_t = pimpl::stored_range_base_t<R>;
37
38 private:
39 template<typename I, typename L>
40 using sized_t = std::conditional_t<RandomIterator<I>, sized_iterator<I>,
41 sized_iterator<I, L>>;
42
43 std::size_t m_how_many = 0;
44
45 template<typename It, typename F, typename L>
46 [[nodiscard]] constexpr It make_begin( F first, L last ) const {
47 if constexpr( RandomIterator<F> ) {
48 auto const range_size = [&] {
49 if constexpr( pimpl::counted_source<F, L> ) {
50 return pimpl::counted_length( first, last );
51 } else {
52 return pimpl::ranges_distance( first, last );
53 }
54 }( );
55 auto const take_size =
56 std::min( { range_size, as<std::ptrdiff_t>( m_how_many ) } );
57 return It{ std::move( first ), as<std::size_t>( take_size ) };
58 } else {
59 return It{ std::move( first ), std::move( last ), m_how_many };
60 }
61 }
62
63 public:
64 using iterator = sized_t<iterator_t<R>, iterator_end_t<R>>;
65 using const_iterator =
66 sized_t<const_iterator_t<R>, const_iterator_end_t<R>>;
67 using iterator_last = sized_iterator_end<iterator_t<R>>;
68 using const_iterator_last = sized_iterator_end<const_iterator_t<R>>;
