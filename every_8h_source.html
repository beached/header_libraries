137 template<Iterator First, Iterator Last>
138 struct every_index_iterator {
139 using iterator_category =
140 common_iterator_category_t<std::forward_iterator_tag,
141 daw::iter_category_t<First>>;
142 using value_type = daw::iter_value_t<First>;
143 using reference = daw::iter_reference_t<First>;
144 using const_reference = daw::iter_const_reference_t<First>;
145 using pointer = daw::iter_pointer_t<First>;
146 using const_pointer = std::remove_pointer_t<pointer> const *;
147 using difference_type = daw::iter_difference_t<First>;
148 using i_am_a_daw_every_iterator_class = void;
149
150 private:
151 First m_first{ };
152 difference_type m_idx = 0; // the selected element we are on
153 difference_type m_count = 0; // how many elements are selected
154 difference_type m_every_nth = 1;
155
156 public:
157 every_index_iterator( ) = default;
158
159 explicit constexpr every_index_iterator(
160 daw::constructible<First> auto &&first,
161 daw::constructible<Last> auto &&last, difference_type every_nth )
162 : m_first( DAW_FWD( first ) )
163 , m_every_nth( every_nth ) {
164 daw_ensure( m_every_nth > 0 );
165 auto const len = as<difference_type>( counted_length( m_first, last ) );
166 m_count = as<difference_type>( len / m_every_nth );
167 if( len % m_every_nth != 0 ) {
168 ++m_count;
169 }
170 }
171
172 [[nodiscard]] constexpr First base( ) const {
173 return m_first + ( m_idx * m_every_nth );
174 }
175
176 [[nodiscard]] constexpr reference operator*( ) {
177 return *base( );
178 }
179
180 [[nodiscard]] constexpr const_reference operator*( ) const {
181 return *base( );
182 }
183
184 [[nodiscard]] constexpr pointer operator->( ) {
185 return std::to_address( base( ) );
186 }
187
188 [[nodiscard]] constexpr const_pointer operator->( ) const {
189 return std::to_address( base( ) );
190 }
191
192 [[nodiscard]] constexpr bool good( ) const {
193 return m_idx < m_count;
194 }
195
196 [[nodiscard]] explicit constexpr operator bool( ) const {
197 return good( );
198 }
199
200 constexpr every_index_iterator &operator++( ) {
201 ++m_idx;
202 return *this;
203 }
204
205 [[nodiscard]] constexpr every_index_iterator operator++( int ) {
206 auto result = *this;
207 operator++( );
208 return result;
209 }
210
211 [[nodiscard]] constexpr bool
212 operator==( every_index_iterator const &rhs ) const {
213 return m_idx == rhs.m_idx and m_first == rhs.m_first;
214 }
215
216 [[nodiscard]] constexpr bool
217 operator==( every_iterator_end<First> const & ) const {
218 return not good( );
219 }
220 };
